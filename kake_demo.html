// Idempotent order event consumer async function handleOrderCreated(event: OrderEvent) { // Idempotency — deduplicate by event ID const processed = await redis.get(`evt:${event.id}`); if (processed) return; await db.$transaction(async (tx) => { // Atomic inventory decrement for (const item of event.lineItems) { await tx.inventory.decrement({ where: { sku: item.sku }, by: item.quantity, }); } await tx.order.update({ where: { id: event.orderId }, data: { status: 'CONFIRMED' } }); }); await redis.setex(`evt:${event.id}`, 86400, '1'); await publishEvent('fulfillment.trigger', event); }

System Architecture
Node.js microservices · React frontend · GraphQL + REST · PostgreSQL + MongoDB + Redis · AWS · CI/CD
⚛️
Frontend — React + TypeScript
  • React 18 with concurrent rendering and Suspense boundaries
  • GraphQL client with normalized cache and optimistic updates
  • Code splitting per route — initial bundle under 80KB gzipped
  • SSR/SSG via Next.js for product pages — Core Web Vitals optimized
  • Cypress E2E tests covering critical checkout and payment flows
⚙️
Backend — Node.js Microservices
  • Order Service — REST API with idempotency keys on all writes
  • Payment Service — Stripe integration with webhook signature verification
  • Inventory Service — atomic decrements, Redis cache, reorder events
  • GraphQL Gateway — versioned schema with cursor-based pagination
  • Event Bus — Redis Streams with at-least-once delivery guarantees
🗄️
Data Layer — Polyglot Persistence
  • PostgreSQL — orders, inventory, pricing, transactions; indexed on hot query paths
  • MongoDB — product content, attributes, media; flexible schema per category
  • Redis — cached reads (p99: 8ms), session storage, event stream, job queues
  • Prisma ORM — type-safe queries, migrations, transaction support
  • Right database for each use case — not one-size-fits-all
☁️
Infrastructure — AWS + CI/CD
  • AWS ECS — containerized Node.js services with auto-scaling
  • CloudFront CDN — edge-cached static assets and SSR responses
  • GitHub Actions — automated test, build, and deploy on every PR merge
  • Datadog — metrics, distributed tracing, structured logs across all services
  • p99 API latency: 42ms · 99.9% uptime SLA maintained

// POST /api/v1/orders — idempotent with optimistic locking router.post('/orders', authenticate, async (req, res) => { const key = req.headers['idempotency-key']; const cached = await redis.get(`idem:${key}`); if (cached) return res.json(JSON.parse(cached)); const order = await db.$transaction(async (tx) => { // Validate and reserve inventory atomically for (const item of req.body.lineItems) { const product = await tx.product.findUniqueOrThrow({ where: { sku: item.sku } }); if (product.inventory < item.quantity) throw new InsufficientStockError(); } return tx.order.create({ data: { ...req.body, status: 'PENDING' } }); }); await publishEvent('order.created', order); await redis.setex(`idem:${key}`, 86400, JSON.stringify(order)); res.status(201).json(order); });
